cmake_minimum_required(VERSION 3.22)
project(copymanga_reader_sr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT EXISTS "${WAIFU2X_SOURCE}/src/waifu2x.cpp")
  message(FATAL_ERROR "Use build.py to prepare the pinned, patched upstream source")
endif()
set(NCNN_VULKAN ON CACHE BOOL "" FORCE)
set(NCNN_SIMPLEVK ON CACHE BOOL "" FORCE)
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
set(NCNN_DISABLE_EXCEPTION OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_INSTALL_SDK OFF CACHE BOOL "" FORCE)
set(NCNN_INT8 OFF CACHE BOOL "" FORCE)
set(NCNN_BF16 OFF CACHE BOOL "" FORCE)
set(NCNN_AVX512 OFF CACHE BOOL "" FORCE)
set(NCNN_XOP OFF CACHE BOOL "" FORCE)
set(NCNN_AVXVNNI OFF CACHE BOOL "" FORCE)
set(NCNN_AVXNECONVERT OFF CACHE BOOL "" FORCE)
file(STRINGS "${WAIFU2X_SOURCE}/src/ncnn/src/CMakeLists.txt" layer_lines REGEX "^ncnn_add_layer\\(")
foreach(line IN LISTS layer_lines)
  string(REGEX MATCH "ncnn_add_layer\\(([A-Za-z0-9_]+)" ignored "${line}")
  string(TOLOWER "${CMAKE_MATCH_1}" layer)
  set(WITH_LAYER_${layer} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(layer convolution crop deconvolution eltwise flatten innerproduct input pooling relu scale split interp packing cast padding gemm)
  set(WITH_LAYER_${layer} ON CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${WAIFU2X_SOURCE}/src/ncnn" ncnn)
foreach(shader waifu2x_preproc waifu2x_postproc waifu2x_preproc_tta waifu2x_postproc_tta)
  execute_process(COMMAND "${CMAKE_COMMAND}"
    "-DSHADER_SRC=${WAIFU2X_SOURCE}/src/${shader}.comp"
    "-DSHADER_COMP_HEADER=${CMAKE_CURRENT_BINARY_DIR}/${shader}.comp.hex.h"
    -P "${WAIFU2X_SOURCE}/src/generate_shader_comp_header.cmake"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
endforeach()
add_library(sr_engine STATIC engine.cpp "${WAIFU2X_SOURCE}/src/waifu2x.cpp")
target_include_directories(sr_engine PRIVATE "${WAIFU2X_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sr_engine PUBLIC ncnn)
if(ANDROID)
  add_library(copymanga_sr SHARED android_jni.cpp)
  target_link_libraries(copymanga_sr PRIVATE sr_engine)
  target_link_options(copymanga_sr PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(copymanga-sr worker.cpp)
  target_link_libraries(copymanga-sr PRIVATE sr_engine)
  if(MINGW)
    target_link_options(copymanga-sr PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()
