cmake_minimum_required(VERSION 3.22)
project(copymanga_restore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
    add_compile_options(/utf-8)
endif()
find_package(Threads REQUIRED)
add_library(restore_core STATIC engine.cpp)
target_include_directories(restore_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE vendor/vulkan/include)
target_link_libraries(restore_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(restore_core PRIVATE /W4)
else()
    target_compile_options(restore_core PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
endif()
add_library(copymanga_restore SHARED jni.cpp)
target_link_libraries(copymanga_restore PRIVATE restore_core)
if(ANDROID)
    target_link_options(copymanga_restore PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_package(JNI REQUIRED COMPONENTS JVM)
    target_include_directories(copymanga_restore PRIVATE ${JNI_INCLUDE_DIRS})
    add_executable(restore-smoke smoke.cpp)
    target_link_libraries(restore-smoke PRIVATE restore_core)
endif()
